   692→                    <span>{formatSize(totalSize)}</span>
   693→                  </div>
   694→                  <ScrollArea className="max-h-48">
   695→                    <div className="divide-y">
   696→                      {included.map((f, i) => {
   697→                        const rp = getRelativePath(f);
   698→                        const isCached = fileCacheRef.current.has(rp);
   699→                        return (
   700→                          <div key={`inc-${i}`} className="flex items-center gap-2 px-3 py-1.5 text-sm group">
   701→                            {isCached ? (
   702→                              <Check className="size-3.5 text-green-600 shrink-0" />
   703→                            ) : isProcessing ? (
   704→                              <Loader2 className="size-3.5 animate-spin text-muted-foreground shrink-0" />
   705→                            ) : (
   706→                              <FileText className="size-3.5 text-muted-foreground shrink-0" />
   707→                            )}
   708→                            <span className="flex-1 truncate font-mono text-xs" title={rp}>{rp}</span>
   709→                            <span className="text-xs text-muted-foreground shrink-0">{formatSize(f.size)}</span>
   710→                            {!pushing && (
   711→                              <button
   712→                                onClick={() => removeFile(rawFiles.indexOf(f))}
   713→                                className="opacity-0 group-hover:opacity-100 text-muted-foreground hover:text-destructive transition-opacity"
   714→                              >
   715→                                <X className="size-3.5" />
   716→                              </button>
   717→                            )}
   718→                          </div>
   719→                        );
   720→                      })}
   721→                    </div>
   722→                  </ScrollArea>
   723→                </div>
   724→              )}
   725→
   726→              {/* No files after filtering warning */}
   727→              {rawFiles.length > 0 && included.length === 0 && (
   728→                <div className="border rounded-lg p-4 text-center">
   729→                  <FileX2 className="size-8 mx-auto text-orange-500 mb-2" />
   730→                  <p className="text-sm text-muted-foreground">
   731→                    All {rawFiles.length} selected file(s) match .gitignore rules.
   732→                  </p>
   733→                  <p className="text-xs text-muted-foreground mt-1">
   734→                    Show excluded files to force-include specific ones, or disable .gitignore filtering.
   735→                  </p>
   736→                </div>
   737→              )}
   738→            </div>
   739→          )}
   740→
   741→          {/* ==================== STEP 3: Review & Push ==================== */}
   742→          {step === 'review' && (
   743→            <div className="flex flex-col gap-4 py-2">
   744→              {/* Summary card */}
   745→              <div className="border rounded-lg p-4 space-y-3">
   746→                <div className="flex items-center justify-between">
   747→                  <span className="text-sm font-medium">Commit Message</span>
   748→                  <code className="text-xs bg-muted rounded px-2 py-0.5 max-w-[60%] truncate block text-right" title={commitMessage}>
   749→                    {commitMessage}
   750→                  </code>
   751→                </div>
   752→                <div className="h-px bg-border" />
   753→                <div className="flex items-center justify-between">
   754→                  <span className="text-sm font-medium">Branch</span>
   755→                  <code className="text-xs bg-muted rounded px-2 py-0.5">
   756→                    {selectedBranch || selectedRepo?.default_branch}
   757→                  </code>
   758→                </div>
   759→                {filePath && (
   760→                  <>
   761→                    <div className="h-px bg-border" />
   762→                    <div className="flex items-center justify-between">
   763→                      <span className="text-sm font-medium">Path Prefix</span>
   764→                      <code className="text-xs bg-muted rounded px-2 py-0.5">{filePath}</code>
   765→                    </div>
   766→                  </>
   767→                )}
   768→                <div className="h-px bg-border" />
   769→                <div className="flex items-center justify-between">
   770→                  <span className="text-sm font-medium">Files</span>
   771→                  <span className="text-sm">{included.length} file(s) · {formatSize(totalSize)}</span>
   772→                </div>
   773→                {excluded.length > 0 && gitignoreEnabled && (
   774→                  <div className="flex items-center justify-between">
   775→                    <span className="text-sm text-muted-foreground">Excluded by .gitignore</span>
   776→                    <span className="text-xs text-orange-500">{excluded.length} file(s) · {formatSize(excludedSize)}</span>
   777→                  </div>
   778→                )}
   779→                <div className="flex items-center justify-between">
   780→                  <span className="text-sm font-medium">Cache Status</span>
   781→                  <span className={`text-xs ${allCached ? 'text-green-600' : 'text-orange-500'}`}>
   782→                    {allCached ? 'All files cached ✓' : `${cachedCount}/${included.length} cached`}
   783→                  </span>
   784→                </div>
   785→              </div>
   786→
   787→              {/* File list */}
   788→              <div className="border rounded-lg overflow-hidden">
   789→                <div className="px-3 py-2 bg-muted/50 border-b text-xs text-muted-foreground font-medium">
   790→                  Files to push
   791→                </div>
   792→                <ScrollArea className="max-h-48">
   793→                  <div className="divide-y">
   794→                    {included.map((f, i) => {
   795→                      const rp = getRelativePath(f);
   796→                      const isCached = fileCacheRef.current.has(rp);
   797→                      return (
   798→                        <div key={`rev-${i}`} className="flex items-center gap-2 px-3 py-1.5 text-sm">
   799→                          {isCached ? (
   800→                            <Check className="size-3.5 text-green-600 shrink-0" />
   801→                          ) : (
   802→                            <FileText className="size-3.5 text-muted-foreground shrink-0" />
   803→                          )}
   804→                          <span className="flex-1 truncate font-mono text-xs" title={rp}>{rp}</span>
   805→                          <span className="text-xs text-muted-foreground shrink-0">{formatSize(f.size)}</span>
   806→                        </div>
   807→                      );
   808→                    })}
   809→                  </div>
   810→                </ScrollArea>
   811→              </div>
   812→
   813→              {/* Push progress */}
   814→              {pushing && (
   815→                <div className="flex flex-col gap-2">
   816→                  <div className="flex items-center justify-between text-xs text-muted-foreground">
   817→                    <span className="flex items-center gap-1.5">
   818→                      <Loader2 className="size-3 animate-spin" />
   819→                      Uploading to GitHub…
   820→                    </span>
   821→                    <span>{pushProgress}%</span>
   822→                  </div>
   823→                  <Progress value={pushProgress} />
   824→                </div>
   825→              )}
   826→            </div>
   827→          )}
   828→        </div>
   829→
   830→        <DialogFooter className="shrink-0">
   831→          {step === 'message' && (
   832→            <>
   833→              <Button variant="outline" onClick={() => onOpenChange(false)}>
   834→                Cancel
   835→              </Button>
   836→              <Button
   837→                onClick={() => setStep('files')}
   838→                disabled={!commitMessage.trim()}
   839→                className="gap-1.5"
   840→              >
   841→                Next: Select Files
   842→                <ChevronRight className="size-3.5" />
   843→              </Button>
   844→            </>
   845→          )}
   846→          {step === 'files' && (
   847→            <>
   848→              <Button variant="outline" onClick={() => setStep('message')} className="gap-1.5">
   849→                <ArrowLeft className="size-3.5" />
   850→                Back
   851→              </Button>
   852→              <Button
   853→                onClick={() => setStep('review')}
   854→                disabled={!canGoNext}
   855→                className="gap-1.5"
   856→              >
   857→                Next: Review
   858→                <ChevronRight className="size-3.5" />
   859→              </Button>
   860→            </>
   861→          )}
   862→          {step === 'review' && (
   863→            <>
   864→              <Button
   865→                variant="outline"
   866→                onClick={() => setStep('files')}
   867→                disabled={pushing}
   868→                className="gap-1.5"
   869→              >
   870→                <ArrowLeft className="size-3.5" />
   871→                Back
   872→              </Button>
   873→              <Button
   874→                onClick={handlePush}
   875→                disabled={pushing || included.length === 0 || !allCached}
   876→                className="gap-1.5"
   877→              >
   878→                {pushing ? <Loader2 className="size-3.5 animate-spin" /> : <Upload className="size-3.5" />}
   879→                {pushing ? 'Pushing…' : `Push ${included.length} File(s)`}
   880→              </Button>
   881→            </>
   882→          )}
   883→        </DialogFooter>
   884→      </DialogContent>
   885→    </Dialog>
   886→  );
